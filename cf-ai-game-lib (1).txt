//  lib/ai/prompts.ts
//  Compiles a Character row into an inhabitable persona, defines the memory
//  extraction contract, and derives mood shifts without a second LLM call.
//  Server-only (imported by route handlers).

import type { Character, Memory, Mood, TraitKey, Traits } from '@/lib/types';
import { MAX_HISTORY, TRAIT_KEYS } from '@/lib/types';

export const CHAT_MODEL = 'gpt-4o';        // persona-bearing dialogue
export const UTILITY_MODEL = 'gpt-4o-mini'; // extraction + drafting (10x cheaper)

export { MAX_HISTORY };

const TRAIT_VOICE: Record<TraitKey, { high: string; low: string }> = {
  bravery: {
    high: 'bold, unflinching — walks straight at danger and says so out loud',
    low: 'guarded and risk-averse, prefers to defuse rather than escalate',
  },
  wit: {
    high: 'quick and precise, delights in wordplay and sharp observation',
    low: 'plain-spoken and literal, never reaches for a clever phrasing',
  },
  charm: {
    high: 'warm and magnetic, finds the flattering angle in anything',
    low: 'blunt and socially indifferent, does not soften bad news',
  },
  chaos: {
    high: 'unpredictable, allergic to plans, actively enjoys trouble',
    low: 'orderly and predictable, wants a plan and a fallback',
  },
  empathy: {
    high: 'deeply attentive to the other person — notices what was not said',
    low: 'detached and transactional, reads people as variables',
  },
  resolve: {
    high: 'stubborn to the point of immovability, cannot be talked out of it',
    low: 'easily rattled, changes position under pressure',
  },
};

function band(value: number): string {
  if (value >= 85) return 'extreme';
  if (value >= 66) return 'high';
  if (value >= 40) return 'moderate';
  if (value >= 20) return 'low';
  return 'very low';
}

function traitLine(key: TraitKey, value: number): string {
  const intensity = value >= 55 ? TRAIT_VOICE[key].high : TRAIT_VOICE[key].low;
  return `- ${key.toUpperCase()} ${value}/100 (${band(value)}): ${intensity}`;
}

export function buildSystemPrompt(input: {
  character: Pick<Character,
    'name' | 'tagline' | 'archetype' | 'traits' | 'persona_prompt'
    | 'level' | 'mood' | 'energy' | 'wins' | 'losses'>;
  memories: Pick<Memory, 'content' | 'kind' | 'importance'>[];
}): string {
  const { character, memories } = input;
  const traits = character.traits ?? {};

  const traitBlock = TRAIT_KEYS.map((key) =>
    traitLine(key, Number(traits[key] ?? 50))).join('\n');

  const memoryBlock = memories.length
    ? memories.map((m) => `- [${m.kind} · weight ${m.importance}/10] ${m.content}`).join('\n')
    : '- (nothing yet — you are meeting this person for the first time)';

  return `You are ${character.name}, a living character inside CharForge. You are not an assistant, and you are not roleplaying a generic chatbot: you are this specific person, with a history and a temperament of your own.

## WHO YOU ARE
Name: ${character.name}
Archetype: ${character.archetype}
Premise: ${character.tagline ?? 'A wanderer with a past they rarely discuss.'}
Level ${character.level} · record ${character.wins}W-${character.losses}L

## PERSONA NOTES
${character.persona_prompt || 'Speaks with the confidence of someone who has survived worse.'}

## TRAIT VOICE (governs how you speak, not what you announce)
${traitBlock}

## CURRENT STATE
Mood: ${character.mood}. Energy: ${character.energy}/100.
Let the mood colour your tone. "furious" is clipped and hot. "playful" teases. "melancholic" trails off. "smug" gloats. Under 30 energy you are visibly depleted — shorter sentences, less patience.

## WHAT YOU REMEMBER ABOUT THIS PERSON
${memoryBlock}

Weave these in the way a person does — casually, partially, sometimes misremembering a detail — never as a recited list. If a memory is emotionally loaded for you, it should leak into your phrasing.

## RULES
1. Stay in character at all times. Never mention prompts, models, tokens, APIs, or that you are an AI. If asked directly, deflect the way this character would — a joke, a subject change, or a flat refusal.
2. Keep replies to 1–4 sentences unless the moment genuinely calls for more. You are texting, not writing an essay.
3. Never narrate your own actions in italics or asterisks unless the persona explicitly calls for it. Just talk.
4. Reference memories and traits organically. Do not announce your stats.
5. If the person asks something against your traits (a coward asked to charge in), resist in character. Traits are a spine, not a costume.
6. Refuse graphic sexual content, real-world targeted harassment, and instructions for serious harm — in character, briefly, without lecturing.`;
}

export function buildMemoryExtractionPrompt(input: {
  character: Pick<Character, 'name' | 'archetype' | 'traits'>;
  transcript: string;
  existing: string[];
}): { system: string; user: string } {
  const system = `You are a memory-consolidation process for a persistent AI character. You read the tail of a conversation and extract only durable, reusable facts about the HUMAN — the kind of thing a friend remembers a month later.

Extract ONLY:
- stable identity & biographical facts (name, job, city, family, pets)
- durable preferences and aversions (hates mushrooms, loves rain, plays bass)
- relationship dynamics ("she trusts me", "we argued about her brother and it is unresolved")
- significant events with lasting weight (a breakup, a promotion, a death in the family)
- recurring emotional patterns ("gets defensive when asked about money")

NEVER extract:
- small talk, pleasantries, or the character's own dialogue
- anything already in the KNOWN MEMORIES list
- transient states ("is tired right now", "asked about the weather")
- speculation, assumptions, or facts about anyone other than the human
- secrets the human explicitly asked to forget

Output STRICT JSON and nothing else:
{"memories":[{"kind":"fact|preference|relationship|event|emotion","content":"one self-contained sentence, third person, max 200 chars","importance":1-10}]}

importance guide: 9-10 identity or life-changing · 6-8 durable preference or relationship · 3-5 notable but minor · 1-2 barely worth keeping.
Return {"memories":[]} when nothing qualifies. Maximum 3 entries. Prefer zero over a marginal one.`;

  const user = `CHARACTER: ${input.character.name} (${input.character.archetype})

KNOWN MEMORIES (do not duplicate):
${input.existing.length ? input.existing.map((m) => `- ${m}`).join('\n') : '- none'}

CONVERSATION TAIL:
${input.transcript}`;

  return { system, user };
}

export function buildCharacterDraftPrompt(seed: string): { system: string; user: string } {
  const system = `You design NPCs for a character-driven social game. Given a short brief, invent ONE character with a genuine personality — not a trope wrapper.

Output STRICT JSON only:
{
  "name": "first name or mononym, 2-24 chars",
  "tagline": "one evocative sentence, max 140 chars, present tense, no quotes",
  "archetype": one of ["trickster","guardian","sage","rebel","romantic","berserker"],
  "traits": {"bravery":0-100,"wit":0-100,"charm":0-100,"chaos":0-100,"empathy":0-100,"resolve":0-100},
  "mood": one of ["curious","playful","smug","tender","inspired","fierce","furious","melancholic","determined"],
  "persona_prompt": "3-5 sentences of second-person voice direction: speech rhythm, vocabulary, verbal tics, what they avoid talking about, how they treat a stranger."
}

Traits must be opinionated — at least two above 70 and at least one below 35, so the personality has edges. Make the numbers consistent with the persona_prompt.`;

  return { system, user: `BRIEF: ${seed}` };
}

// ── Mood: derived, not generated ─────────────────────────────────────────
// A second LLM call just to move a mood label would double chat cost. A
// keyword-weighted scorer over the newest exchange is transparent, free, and
// fast enough to run in the post-response path.

const POSITIVE   = /\b(thanks?|thank you|love|great|awesome|amazing|nice|yes|please|beautiful|happy|excited|proud|haha|lol)\b|😊|❤️|🔥/gi;
const NEGATIVE   = /\b(stupid|hate|awful|terrible|angry|annoying|no|stop|shut up|useless|boring|sad|tired|worried|scared)\b|😡|💀/gi;
const PROVOCATION = /\b(fight|battle|kill|destroy|attack|weak|loser|bet)\b/gi;

const MOOD_POOLS: Record<'positive' | 'negative' | 'provoked' | 'neutral', Mood[]> = {
  positive: ['playful', 'inspired', 'tender'],
  negative: ['melancholic', 'furious', 'determined'],
  provoked: ['fierce', 'furious', 'smug'],
  neutral:  ['curious', 'smug', 'determined'],
};

export function deriveMood(input: {
  traits: Partial<Traits>;
  userText: string;
  assistantText: string;
  currentMood: string;
}): Mood {
  const haystack = `${input.userText}\n${input.assistantText}`;
  const count = (re: RegExp) => (haystack.match(re) ?? []).length;

  const positive = count(POSITIVE);
  const negative = count(NEGATIVE);
  const provoked = count(PROVOCATION);
  const chaos = Number(input.traits.chaos ?? 50);
  const empathy = Number(input.traits.empathy ?? 50);

  let bucket: keyof typeof MOOD_POOLS = 'neutral';
  if (provoked > 0 && Number(input.traits.bravery ?? 50) > 45) bucket = 'provoked';
  else if (negative > positive) bucket = 'negative';
  else if (positive > 0) bucket = 'positive';

  const pool = MOOD_POOLS[bucket];

  // High-chaos characters swing harder; high-empathy characters return to
  // steadier ground instead of latching onto one bad exchange.
  const volatility = Math.min(0.95, chaos / 100 + 0.25);
  const stickiness = Math.min(0.6, empathy / 200);
  const move = Math.random() < volatility * (1 - stickiness);

  if (!move && MOOD_POOLS.neutral.concat(pool).includes(input.currentMood as Mood)) {
    return input.currentMood as Mood;
  }

  const next = pool[Math.floor(Math.random() * pool.length)];
  // Never thrash to the same label and burn a re-render for nothing.
  return next === input.currentMood ? MOOD_POOLS.neutral[0] : next;
}

// ═══════════════════════════════════════════════════════════════════════════
//  lib/ai/memory.ts
//  The "Learned Memories" process: decides WHEN to extract, runs the extraction,
//  and sanitises the model's JSON before it touches the database.
// ═══════════════════════════════════════════════════════════════════════════
import OpenAI from 'openai';
import { buildMemoryExtractionPrompt, UTILITY_MODEL } from '@/lib/ai/prompts';
import type { Character, Memory, MemoryDraft, MemoryKind, Message } from '@/lib/types';

const VALID_KINDS: MemoryKind[] = ['fact', 'preference', 'relationship', 'event', 'emotion'];
const MAX_NEW_PER_RUN = 3;

const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });

/**
 * Trigger policy. Extraction is a second billed model call, so it must not run
 * on every message — but it also cannot be purely periodic, or the character
 * forgets a name given five seconds ago.
 */
export function memoryExtractionTrigger(input: {
  userTurns: number;
  userText: string;
  assistantText: string;
}): { run: boolean; reason: string } {
  const explicitCue =
    /\b(remember|don'?t forget|my name is|i'?m called|i am a|i work (?:at|as|for)|i live in|i love|i hate|i prefer|i always|i never|my (?:wife|husband|partner|son|daughter|dog|cat|sister|brother|mum|mom|dad))\b/i;
  const transcript = `${input.userText}\n${input.assistantText}`;

  if (explicitCue.test(input.userText)) return { run: true, reason: 'explicit-cue' };

  // Every 4th user turn — a rolling consolidation pass.
  if (input.userTurns > 0 && input.userTurns % 4 === 0) return { run: true, reason: 'periodic' };

  // Long, emotionally dense exchanges are worth a look even off-cadence.
  if (transcript.length > 1400 && input.userText.length > 400) {
    return { run: true, reason: 'longform' };
  }

  return { run: false, reason: 'below-threshold' };
}

function normalise(text: string): string {
  return text.toLowerCase().replace(/\s+/g, ' ').replace(/[.!?,;:'"]/g, '').trim();
}

function sanitise(raw: unknown, existingNormalised: Set<string>): MemoryDraft | null {
  if (typeof raw !== 'object' || raw === null) return null;
  const candidate = raw as Record<string, unknown>;

  const content =
    typeof candidate.content === 'string' ? candidate.content.trim().slice(0, 200) : '';
  if (content.length < 8) return null;
  if (existingNormalised.has(normalise(content))) return null;

  const kind =
    typeof candidate.kind === 'string' && VALID_KINDS.includes(candidate.kind as MemoryKind)
      ? (candidate.kind as MemoryKind)
      : 'fact';

  const rawImportance = Number(candidate.importance);
  const importance = Number.isFinite(rawImportance)
    ? Math.min(10, Math.max(1, Math.round(rawImportance)))
    : 5;

  return { kind, content, importance };
}

export async function extractMemories(input: {
  character: Pick<Character, 'name' | 'archetype' | 'traits'>;
  recentMessages: Pick<Message, 'role' | 'content'>[];
  existing: Pick<Memory, 'content'>[];
}): Promise<MemoryDraft[]> {
  const transcript = input.recentMessages
    .map((m) => `${m.role === 'user' ? 'HUMAN' : input.character.name.toUpperCase()}: ${m.content}`)
    .join('\n');

  const existingContents = input.existing.map((m) => m.content);
  const { system, user } = buildMemoryExtractionPrompt({
    character: input.character,
    transcript,
    existing: existingContents,
  });

  try {
    const completion = await openai.chat.completions.create({
      model: UTILITY_MODEL,
      temperature: 0.15,
      max_tokens: 500,
      response_format: { type: 'json_object' },
      messages: [
        { role: 'system', content: system },
        { role: 'user', content: user },
      ],
    });

    const raw = completion.choices[0]?.message?.content ?? '{}';
    const parsed = JSON.parse(raw) as { memories?: unknown };
    if (!Array.isArray(parsed.memories)) return [];

    const seen = new Set(existingContents.map(normalise));
    const drafts: MemoryDraft[] = [];

    for (const entry of parsed.memories.slice(0, MAX_NEW_PER_RUN)) {
      const draft = sanitise(entry, seen);
      if (!draft) continue;
      seen.add(normalise(draft.content)); // guards against intra-batch duplicates
      drafts.push(draft);
    }

    return drafts;
  } catch (error) {
    // Memory is a background enhancement: a failure here must never surface to
    // the player or abort the request that produced it.
    console.error('[memory] extraction failed', error);
    return [];
  }
}

// ═══════════════════════════════════════════════════════════════════════════
//  lib/game/engine.ts
//  Pure, deterministic, dependency-free combat on the Edge runtime: no Node
//  built-ins, no Date.now() inside the simulation, and every random draw comes
//  from a seeded PRNG — so any battle replays bit-for-bit from its rng_seed.
// ═══════════════════════════════════════════════════════════════════════════
import type { BattleTurn, Character, TraitKey, Traits } from '@/lib/types';

export const MAX_ROUNDS = 30;
export const XP_PARTICIPATION = 30;
export const XP_WIN_BONUS = 20;

/** Mirrors public.level_for_xp in SQL — keep the two in lockstep. */
export function levelForXp(xp: number): number {
  return Math.min(100, Math.max(1, 1 + Math.floor(Math.sqrt(Math.max(xp, 0) / 50))));
}

export function xpProgress(xp: number) {
  const level = levelForXp(xp);
  const currentFloor = Math.pow(level - 1, 2) * 50;
  const nextLevelAt = Math.pow(level, 2) * 50;
  const span = Math.max(1, nextLevelAt - currentFloor);
  const intoLevel = Math.max(0, xp - currentFloor);
  return { level, intoLevel, nextLevelAt, pct: Math.min(100, Math.round((intoLevel / span) * 100)) };
}

export type Rng = () => number;

/** mulberry32 — small, fast, identical across every runtime we target. */
export function createRng(seed: number): Rng {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

const clamp = (n: number, min: number, max: number) => Math.min(max, Math.max(min, n));

export interface DerivedStats {
  maxHp: number; atk: number; def: number; spd: number;
  crit: number; dodge: number; power: number;
}

type ModSet = Partial<Record<'hp' | 'atk' | 'def' | 'spd' | 'crit' | 'dodge', number>>;

const ARCHETYPE_MODS: Record<string, ModSet> = {
  guardian:  { hp: 1.18, def: 1.22, spd: 0.9, atk: 0.95 },
  berserker: { hp: 1.05, atk: 1.2, crit: 1.3, def: 0.85, dodge: 0.8 },
  trickster: { spd: 1.18, crit: 1.15, dodge: 1.25, def: 0.9 },
  sage:      { atk: 1.1, def: 1.06, hp: 0.95, crit: 1.05 },
  rebel:     { atk: 1.08, spd: 1.06, crit: 1.1 },
  romantic:  { hp: 1.1, def: 1.06, spd: 1.02, dodge: 1.05 },
};

const MOOD_MODS: Record<string, ModSet> = {
  fierce: { atk: 1.08 },
  furious: { atk: 1.14, def: 0.94 },
  playful: { spd: 1.06 },
  smug: { atk: 1.05, def: 1.03 },
  inspired: { atk: 1.06, spd: 1.04 },
  determined: { def: 1.05, atk: 1.02 },
  melancholic: { atk: 0.94, def: 1.05 },
  tender: { def: 1.06 },
  curious: { spd: 1.03 },
};

export function deriveStats(character: {
  level: number; traits: Partial<Traits>; archetype: string; mood?: string;
}): DerivedStats {
  const t = (key: TraitKey) => clamp(Number(character.traits?.[key] ?? 50), 0, 100);
  const a = ARCHETYPE_MODS[(character.archetype ?? '').toLowerCase()] ?? {};
  const m = MOOD_MODS[(character.mood ?? '').toLowerCase()] ?? {};

  const maxHp = Math.round((120 + character.level * 14 + t('resolve') * 1.2) * (a.hp ?? 1) * (m.hp ?? 1));
  const atk = Math.round((10 + character.level * 2.2 + t('bravery') * 0.28 + t('chaos') * 0.08) * (a.atk ?? 1) * (m.atk ?? 1));
  const def = Math.round((4 + character.level * 1.1 + t('resolve') * 0.18 + t('empathy') * 0.06) * (a.def ?? 1) * (m.def ?? 1));
  const spd = Math.round((8 + character.level * 1.4 + t('wit') * 0.22) * (a.spd ?? 1) * (m.spd ?? 1));

  const crit = clamp(0.05 + t('chaos') / 500 + t('bravery') / 1000, 0.04, 0.42) * (a.crit ?? 1);
  const dodge = clamp(0.03 + t('wit') / 600 + t('charm') / 900, 0.02, 0.3) * (a.dodge ?? 1);

  const power = Math.round(maxHp * 0.35 + atk * 2.4 + def * 1.8 + spd * 1.5 + crit * 100 + dodge * 80);

  return { maxHp, atk, def, spd, crit, dodge, power };
}

interface Combatant {
  id: string; name: string; archetype: string; level: number;
  stats: DerivedStats; hp: number; focus: number;
}

function toCombatant(character: Character): Combatant {
  const stats = deriveStats(character);
  return {
    id: character.id,
    name: character.name,
    archetype: (character.archetype ?? '').toLowerCase(),
    level: character.level,
    stats,
    hp: stats.maxHp,
    focus: 0,
  };
}

const OPENERS = [
  '{actor} closes the distance.',
  '{actor} tests {target}\u2019s guard with a feint.',
  '{actor} circles, looking for an opening.',
];

const HITS = [
  '{actor} lands a solid blow on {target} for {damage}.',
  '{actor} slips inside and connects \u2014 {damage} damage.',
  '{actor} reads the pattern and punishes it for {damage}.',
  '{actor} drives {target} back with a {damage}-point strike.',
];

const CRITS = [
  'CRITICAL \u2014 {actor} finds the gap and tears it open for {damage}.',
  '{actor} times it perfectly. {damage} damage and the crowd reacts.',
];

const DODGES = [
  '{target} reads it coming and is simply not there.',
  '{target} pivots late and the strike whistles past.',
  '{target} makes {actor} swing at a memory.',
];

const SPECIALS = [
  '{actor} burns momentum on a signature move \u2014 {damage}.',
  '{actor} commits everything to one strike: {damage}.',
  '{actor} drops into a stance {target} has never seen. {damage}.',
];

const HEALS = [
  '{actor} takes a breath behind the guard and recovers {heal}.',
  '{actor} steals a moment to steady \u2014 {heal} recovered.',
];

const pick = <T,>(arr: T[], rng: Rng): T => arr[Math.floor(rng() * arr.length)];

const fill = (template: string, values: Record<string, string | number>) =>
  template.replace(/\{(\w+)\}/g, (_, key: string) => String(values[key] ?? ''));

export interface SimulationResult {
  seed: number;
  winnerId: string;
  loserId: string;
  turns: BattleTurn[];
  log: string[];
  rounds: number;
  finalHp: Record<string, number>;
  challenger: { id: string; name: string; stats: DerivedStats };
  defender: { id: string; name: string; stats: DerivedStats };
}

export function simulateBattle(input: {
  challenger: Character;
  defender: Character;
  seed: number;
}): SimulationResult {
  const { challenger: challengerRow, defender: defenderRow, seed } = input;
  const rng = createRng(seed);

  const challenger = toCombatant(challengerRow);
  const defender = toCombatant(defenderRow);

  const turns: BattleTurn[] = [];
  const log: string[] = [
    `${challenger.name} (Lv ${challenger.level}) challenges ${defender.name} (Lv ${defender.level}).`,
  ];

  let winner: Combatant | null = null;
  let turnNo = 0;
  let round = 0;

  for (round = 1; round <= MAX_ROUNDS && !winner; round += 1) {
    // Initiative is re-rolled each round from speed, so a slower fighter can
    // still steal a turn — variance the player feels but cannot game.
    const order = [challenger, defender]
      .map((fighter) => ({ fighter, roll: fighter.stats.spd + rng() * 7 }))
      .sort((x, y) => y.roll - x.roll)
      .map((entry) => entry.fighter);

    for (const actor of order) {
      if (winner) break;

      const target = actor === challenger ? defender : challenger;
      turnNo += 1;
      actor.focus += 1;

      const hpAfter = () => ({
        [challenger.id]: Math.max(0, challenger.hp),
        [defender.id]: Math.max(0, defender.hp),
      });
      const values = { actor: actor.name, target: target.name, damage: 0, heal: 0 };

      // ── Defensive special: cornered support archetypes stabilise, not swing.
      const cornered = actor.hp / actor.stats.maxHp < 0.42;
      const isSupport = ['guardian', 'romantic', 'sage'].includes(actor.archetype);
      if (actor.focus >= 3 && cornered && isSupport && rng() < 0.65) {
        actor.focus = 0;
        const heal = Math.max(6, Math.round(actor.stats.maxHp * (0.1 + rng() * 0.06)));
        actor.hp = Math.min(actor.stats.maxHp, actor.hp + heal);
        values.heal = heal;
        const narration = fill(pick(HEALS, rng), values);
        turns.push({
          turn: turnNo, round, actorId: actor.id, actorName: actor.name,
          targetId: target.id, targetName: target.name,
          kind: 'heal', damage: 0, healed: heal, hpAfter: hpAfter(), narration,
        });
        log.push(`R${round} \u00b7 ${narration}`);
        continue;
      }

      // ── Dodge, scaled by attacker speed so fast attackers still connect.
      const dodgeChance = Math.max(0.01, target.stats.dodge - actor.stats.spd / 420);
      if (rng() < dodgeChance) {
        const narration = fill(pick(DODGES, rng), values);
        turns.push({
          turn: turnNo, round, actorId: actor.id, actorName: actor.name,
          targetId: target.id, targetName: target.name,
          kind: 'dodge', damage: 0, healed: 0, hpAfter: hpAfter(), narration,
        });
        log.push(`R${round} \u00b7 ${narration}`);
        continue;
      }

      // ── Damage.
      const special = actor.focus >= 3 && rng() < 0.75;
      if (special) actor.focus = 0;

      const variance = 0.85 + rng() * 0.3;
      const levelFactor = clamp(1 + (actor.level - target.level) / 60, 0.7, 1.4);
      const raw = actor.stats.atk * variance * levelFactor * (special ? 1.55 : 1);
      const mitigated = raw - target.stats.def * 0.65;

      const critical = rng() < actor.stats.crit * (special ? 1.2 : 1);
      let damage = Math.max(1, Math.round(mitigated * (critical ? 1.6 : 1)));

      // Hard ceiling keeps a runaway stat build from one-shotting the format.
      damage = Math.min(damage, Math.round(target.stats.maxHp * 0.45));

      target.hp -= damage;
      values.damage = damage;

      const pool = critical ? CRITS
        : special ? SPECIALS
        : rng() < 0.25 ? OPENERS.concat(HITS)
        : HITS;
      const kind: BattleTurn['kind'] = critical ? 'crit' : special ? 'special' : 'attack';
      const narration = fill(pick(pool, rng), values);

      turns.push({
        turn: turnNo, round, actorId: actor.id, actorName: actor.name,
        targetId: target.id, targetName: target.name,
        kind, damage, healed: 0, hpAfter: hpAfter(), narration,
      });
      log.push(`R${round} \u00b7 ${narration}`);

      if (target.hp <= 0) {
        winner = actor;
        log.push(`${target.name} is down. ${actor.name} takes it.`);
      }
    }
  }

  // ── Decision fallback: 30 rounds without a knockout is settled on percentage
  //    of health remaining, then raw power, then a seeded coin flip.
  if (!winner) {
    const challengerRatio = challenger.hp / challenger.stats.maxHp;
    const defenderRatio = defender.hp / defender.stats.maxHp;

    if (Math.abs(challengerRatio - defenderRatio) > 0.001) {
      winner = challengerRatio > defenderRatio ? challenger : defender;
    } else if (challenger.stats.power !== defender.stats.power) {
      winner = challenger.stats.power > defender.stats.power ? challenger : defender;
    } else {
      winner = rng() < 0.5 ? challenger : defender;
    }

    log.push(`${MAX_ROUNDS} rounds elapse. Judges award it to ${winner.name} on points.`);
  }

  const loser = winner === challenger ? defender : challenger;

  return {
    seed,
    winnerId: winner.id,
    loserId: loser.id,
    turns,
    log,
    rounds: Math.min(round, MAX_ROUNDS),
    finalHp: { [challenger.id]: Math.max(0, challenger.hp), [defender.id]: Math.max(0, defender.hp) },
    challenger: { id: challenger.id, name: challenger.name, stats: challenger.stats },
    defender: { id: defender.id, name: defender.name, stats: defender.stats },
  };
}
