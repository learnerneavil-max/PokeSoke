```ts
// ═══════════════════════════════════════════════════════════════════════════
//  app/api/chat/route.ts
//
//  Streaming LLM chat on the Vercel Edge runtime, with post-response memory
//  extraction and progression handled by `after()`.
//
//  RUNTIME NOTE
//  Edge gives us ~50ms cold starts instead of ~500ms, which matters when the
//  first token is the whole experience. The trade-off is a platform wall-clock
//  budget (≈25s on Hobby) rather than Vercel's configurable `maxDuration`.
//  GPT-4o at max_tokens 500 finishes well inside that. If you later generate
//  long-form output (>30s), flip to:
//      export const runtime = 'nodejs';
//      export const maxDuration = 60;
//  Nothing else in this file has to change.
// ═══════════════════════════════════════════════════════════════════════════
import { NextResponse, type NextRequest } from 'next/server';
import { after } from 'next/server';
import OpenAI from 'openai';
import { buildSystemPrompt, CHAT_MODEL, deriveMood } from '@/lib/ai/prompts';
import { extractMemories, memoryExtractionTrigger, MEMORY_WINDOW } from '@/lib/ai/memory';
import { levelForXp, xpProgress } from '@/lib/game/engine';
import { createSupabaseServerClient, createSupabaseTokenClient } from '@/lib/supabase/server';
import { MAX_HISTORY, type Character, type Memory, type Message } from '@/lib/types';

export const runtime = 'edge';
export const dynamic = 'force-dynamic';
export const maxDuration = 30;

const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });

const MAX_MESSAGE_CHARS = 4000;
const RATE_LIMIT_PER_MINUTE = 20;
const XP_PER_MESSAGE = 6;
const XP_PER_BONUS_MAX = 6;

const json = (body: unknown, status = 200) =>
  NextResponse.json(body, { status, headers: { 'Cache-Control': 'no-store' } });

interface StreamOutcome {
  text: string;
  aborted: boolean;
}

// ─────────────────────────────────────────────────────────────────────────
//  GET /api/chat?characterId=…  →  { character, messages, memories }
//  Hydrates the chat panel for one character.
// ─────────────────────────────────────────────────────────────────────────
export async function GET(request: NextRequest) {
  const characterId = request.nextUrl.searchParams.get('characterId');
  if (!characterId) return json({ error: 'characterId is required' }, 400);

  const supabase = await createSupabaseServerClient();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return json({ error: 'Unauthorized' }, 401);

  const { data: character } = await supabase
    .from('characters')
    .select('*')
    .eq('id', characterId)
    .eq('owner_id', user.id)
    .maybeSingle();

  if (!character) return json({ error: 'Character not found' }, 404);

  const [messages, memories] = await Promise.all([
    supabase
      .from('messages')
      .select('*')
      .eq('character_id', characterId)
      .order('created_at', { ascending: false })
      .limit(60),
    supabase
      .from('memories')
      .select('*')
      .eq('character_id', characterId)
      .order('importance', { ascending: false })
      .order('created_at', { ascending: false })
      .limit(MEMORY_WINDOW),
  ]);

  return json({
    character,
    messages: (messages.data ?? []).slice().reverse(),
    memories: memories.data ?? [],
  });
}

// ─────────────────────────────────────────────────────────────────────────
//  POST /api/chat   { characterId, message }
//  Responds with text/event-stream:
//    event: meta   { userMessageId, assistantMessageId, characterId }
//    event: delta  { t: "..." }
//    event: done   { ok: true }
//    event: error  { message }
// ─────────────────────────────────────────────────────────────────────────
export async function POST(request: NextRequest) {
  // ── 1. Validate input before touching auth or the model.
  let payload: unknown;
  try {
    payload = await request.json();
  } catch {
    return json({ error: 'Request body must be JSON.' }, 400);
  }

  const { characterId, message } = (payload ?? {}) as { characterId?: string; message?: string };
  const trimmed = typeof message === 'string' ? message.trim() : '';

  if (!characterId || typeof characterId !== 'string') {
    return json({ error: 'characterId is required.' }, 400);
  }
  if (trimmed.length === 0 || trimmed.length > MAX_MESSAGE_CHARS) {
    return json({ error: `message must be between 1 and ${MAX_MESSAGE_CHARS} characters.` }, 400);
  }

  // ── 2. Identify the caller. getUser() verifies the JWT with Supabase Auth;
  //      getSession() is only used to lift the raw token for the after() work.
  const supabase = await createSupabaseServerClient();
  const { data: { user }, error: authError } = await supabase.auth.getUser();
  if (authError || !user) return json({ error: 'Unauthorized' }, 401);

  const { data: { session } } = await supabase.auth.getSession();
  const accessToken = session?.access_token;
  if (!accessToken) return json({ error: 'Session expired. Please sign in again.' }, 401);

  // ── 3. Load the character, its memory bank, and the recent transcript.
  const { data: character } = await supabase
    .from('characters')
    .select('*')
    .eq('id', characterId)
    .eq('owner_id', user.id)
    .maybeSingle();

  if (!character) return json({ error: 'Character not found.' }, 404);

  const since = new Date(Date.now() - 60_000).toISOString();

  const [memoriesResult, historyResult, userTurnCountResult, recentCountResult] = await Promise.all([
    supabase
      .from('memories')
      .select('*')
      .eq('character_id', characterId)
      .order('importance', { ascending: false })
      .order('created_at', { ascending: false })
      .limit(MEMORY_WINDOW),
    supabase
      .from('messages')
      .select('*')
      .eq('character_id', characterId)
      .order('created_at', { ascending: false })
      .limit(MAX_HISTORY),
    supabase
      .from('messages')
      .select('id', { count: 'exact', head: true })
      .eq('character_id', characterId)
      .eq('role', 'user'),
    supabase
      .from('messages')
      .select('id', { count: 'exact', head: true })
      .eq('character_id', characterId)
      .eq('role', 'user')
      .gte('created_at', since),
  ]);

  // ── 4. Throttle. Counting rows keeps this stateless across edge isolates,
  //      unlike an in-memory limiter which resets on every cold start.
  if ((recentCountResult.count ?? 0) >= RATE_LIMIT_PER_MINUTE) {
    return json({ error: 'Slow down — that is a lot of talking. Try again in a minute.' }, 429);
  }

  const memories = (memoriesResult.data ?? []) as Memory[];
  const history = ((historyResult.data ?? []) as Message[]).slice().reverse();
  const userTurns = userTurnCountResult.count ?? 0;

  // ── 5. Persist the user's turn immediately, so a dropped connection never
  //      loses their message. The assistant's turn is written after the stream.
  const userMessageId = crypto.randomUUID();
  const assistantMessageId = crypto.randomUUID();

  const { error: insertError } = await supabase.from('messages').insert({
    id: userMessageId,
    character_id: characterId,
    owner_id: user.id,
    role: 'user',
    content: trimmed,
  });

  if (insertError) {
    console.error('[chat] failed to persist user message', insertError);
    return json({ error: 'Could not save your message. Try again.' }, 500);
  }

  // ── 6. Compile the persona and open the completion.
  const systemPrompt = buildSystemPrompt({ character: character as Character, memories });

  const chatMessages: OpenAI.Chat.Completions.ChatCompletionMessageParam[] = [
    { role: 'system', content: systemPrompt },
    ...history.map((m) => ({ role: m.role as 'user' | 'assistant', content: m.content })),
    { role: 'user', content: trimmed },
  ];

  let completion: Awaited<ReturnType<typeof openai.chat.completions.create>>;
  try {
    completion = await openai.chat.completions.create(
      {
        model: CHAT_MODEL,
        temperature: 0.9,
        presence_penalty: 0.35,
        frequency_penalty: 0.2,
        max_tokens: 500,
        stream: true,
        messages: chatMessages,
      },
      // Forwarding the client's abort signal stops billing the moment a user
      // navigates away or closes the tab mid-generation.
      { signal: request.signal },
    );
  } catch (error) {
    console.error('[chat] completion failed to open', error);
    return json({ error: 'The character is not responding right now. Try again.' }, 502);
  }

  // ── 7. Deferred completion channel.
  //      `after()` must be registered synchronously inside the request scope,
  //      so we hand it a promise that the stream resolves when it finishes
  //      (or when the client disconnects, whichever comes first).
  let settle!: (outcome: StreamOutcome) => void;
  const streamFinished = new Promise<StreamOutcome>((resolve) => {
    settle = resolve;
  });

  after(async () => {
    const outcome = await streamFinished;
    const db = createSupabaseTokenClient(accessToken);

    try {
      const replyText = outcome.text.trim();

      // 7a. Persist the assistant turn (partial text if the client bailed).
      if (replyText.length > 0) {
        await db.from('messages').insert({
          id: assistantMessageId,
          character_id: characterId,
          owner_id: user.id,
          role: 'assistant',
          content: replyText.slice(0, 8000),
        });
      }

      // 7b. Memory extraction — the "Learned Memories" pipeline.
      if (replyText.length > 0) {
        const trigger = memoryExtractionTrigger({ userTurns, userText: trimmed, assistantText: replyText });
        console.log(`[memory] trigger=${trigger.run} reason=${trigger.reason}`);

        if (trigger.run) {
          const transcriptTail = history.slice(-6).map((m) => ({ role: m.role, content: m.content }));
          const drafts = await extractMemories({
            character: character as Character,
            recentMessages: [
              ...transcriptTail,
              { role: 'user', content: trimmed },
              { role: 'assistant', content: replyText },
            ],
            existing: memories.map((m) => ({ content: m.content })),
          });

          if (drafts.length > 0) {
            // The unique (character_id, content) constraint makes this a no-op
            // if the model re-derives something already stored.
            await db.from('memories').upsert(
              drafts.map((draft) => ({
                character_id: characterId,
                owner_id: user.id,
                kind: draft.kind,
                content: draft.content,
                importance: draft.importance,
              })),
              { onConflict: 'character_id,content', ignoreDuplicates: true },
            );
          }
        }
      }

      // 7c. Progression: XP for the exchange, then a derived mood swing.
      const bonus = Math.min(XP_PER_BONUS_MAX, Math.floor(trimmed.length / 160));
      const xpGain = XP_PER_MESSAGE + bonus;
      const nextXp = character.xp + xpGain;

      const nextMood = deriveMood({
        traits: character.traits ?? {},
        userText: trimmed,
        assistantText: replyText,
        currentMood: character.mood,
      });

      await db
        .from('characters')
        .update({
          xp: nextXp,
          level: levelForXp(nextXp),
          mood: nextMood,
          energy: Math.min(100, Math.max(0, character.energy - 1)),
        })
        .eq('id', characterId);
    } catch (error) {
      // A failure in the background path must not look like a failed chat.
      console.error('[chat] post-response work failed', error);
    }
  });

  // ── 8. The stream itself.
  const encoder = new TextEncoder();
  let accumulated = '';
  let clientAborted = request.signal.aborted;

  const body = new ReadableStream<Uint8Array>({
    async start(controller) {
      const send = (event: string, data: unknown) => {
        controller.enqueue(encoder.encode(`event: ${event}\ndata: ${JSON.stringify(data)}\n\n`));
      };

      send('meta', { userMessageId, assistantMessageId, characterId });

      try {
        for await (const chunk of completion) {
          const delta = chunk.choices[0]?.delta?.content;
          if (!delta) continue;
          accumulated += delta;
          send('delta', { t: delta });
        }
        send('done', { ok: true });
      } catch (error) {
        if (request.signal.aborted) {
          clientAborted = true;
        } else {
          console.error('[chat] stream error', error);
          try {
            send('error', { message: 'The character lost their train of thought. Send that again.' });
          } catch {
            // Controller already closed — nothing left to tell the client.
          }
        }
      } finally {
        try {
          controller.close();
        } catch {
          // Already closed.
        }
        settle({ text: accumulated, aborted: clientAborted });
      }
    },

    cancel() {
      // Reader walked away. Keep whatever tokens already arrived.
      clientAborted = true;
      settle({ text: accumulated, aborted: true });
    },
  });

  return new Response(body, {
    headers: {
      'Content-Type': 'text/event-stream; charset=utf-8',
      // no-transform is load-bearing: without it, some CDNs buffer the whole
      // response and the typewriter effect disappears.
      'Cache-Control': 'no-cache, no-store, no-transform',
      Connection: 'keep-alive',
      'X-Accel-Buffering': 'no',
    },
  });
}

// Re-exported for the client so it can render the same progress bar without
// duplicating the XP curve.
export { xpProgress };
```
