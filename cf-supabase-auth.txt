```ts
//  lib/supabase/client.ts  —  browser client (singleton)
// ═══════════════════════════════════════════════════════════════════════════
import { createBrowserClient } from '@supabase/ssr';
import type { Database } from '@/lib/types';

type BrowserClient = ReturnType<typeof createBrowserClient<Database>>;

let cached: BrowserClient | null = null;

/**
 * One instance per browser tab. Creating multiple GoTrue clients causes
 * auth-lock contention (dropped refreshes → random logouts) and duplicates
 * every Realtime socket, so this is memoised deliberately.
 */
export function createSupabaseBrowserClient(): BrowserClient {
  if (cached) return cached;

  cached = createBrowserClient<Database>(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
  );

  return cached;
}
```

```ts
// ═══════════════════════════════════════════════════════════════════════════
//  lib/supabase/server.ts  —  three server clients, three trust levels
// ═══════════════════════════════════════════════════════════════════════════
import { createServerClient } from '@supabase/ssr';
import { createClient } from '@supabase/supabase-js';
import { cookies } from 'next/headers';
import type { Database } from '@/lib/types';

const SUPABASE_URL = process.env.NEXT_PUBLIC_SUPABASE_URL!;
const SUPABASE_ANON_KEY = process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!;

/**
 * Request-scoped SSR client. Resolves identity from the auth cookies and
 * respects RLS as that user. `cookies()` is async in Next 15 — do not cache
 * the resolved store at module scope or every request shares one session.
 */
export async function createSupabaseServerClient() {
  const cookieStore = await cookies();

  return createServerClient<Database>(SUPABASE_URL, SUPABASE_ANON_KEY, {
    cookies: {
      getAll() {
        return cookieStore.getAll();
      },
      setAll(cookiesToSet) {
        try {
          for (const { name, value, options } of cookiesToSet) {
            cookieStore.set(name, value, options);
          }
        } catch {
          // Thrown when called from a Server Component, which cannot write
          // headers. Safe to swallow only because middleware.ts refreshes
          // the session on every navigation.
        }
      },
    },
  });
}

/**
 * RLS-scoped client bound to a captured access token instead of cookies.
 *
 * Use this inside `after()`: once the response has been flushed, the Next.js
 * request scope (and its cookie store) is no longer guaranteed readable, but
 * a plain bearer token keeps working. RLS still applies as that user, so this
 * is not a privilege escalation — it is the same identity, minus the cookie
 * dependency.
 */
export function createSupabaseTokenClient(accessToken: string) {
  return createClient<Database>(SUPABASE_URL, SUPABASE_ANON_KEY, {
    auth: {
      persistSession: false,
      autoRefreshToken: false,
      detectSessionInUrl: false,
      storageKey: 'cf-token-scoped',
    },
    global: { headers: { Authorization: `Bearer ${accessToken}` } },
  });
}

/**
 * Bypasses RLS entirely. Only for work the user must not be trusted to do —
 * here, publishing Realtime broadcasts on their behalf. Never import this
 * from a Client Component or a file that the browser bundle can reach.
 */
export function createSupabaseAdminClient() {
  const secret = process.env.SUPABASE_SERVICE_ROLE_KEY;
  if (!secret) {
    throw new Error('SUPABASE_SERVICE_ROLE_KEY is not set — cannot create an admin client.');
  }

  return createClient<Database>(SUPABASE_URL, secret, {
    auth: {
      persistSession: false,
      autoRefreshToken: false,
      detectSessionInUrl: false,
      storageKey: 'cf-admin',
    },
  });
}
```

```ts
// ═══════════════════════════════════════════════════════════════════════════
//  middleware.ts  —  project root
//  Refreshes the Supabase session on every navigation and writes the rotated
//  cookies back to both the request and the response.
//
//  NOTE: `/api` is deliberately excluded from the matcher. The AI routes do
//  their own `auth.getUser()` check, and a middleware round-trip to Supabase
//  Auth in front of a streaming route adds latency to time-to-first-token.
// ═══════════════════════════════════════════════════════════════════════════
import { NextResponse, type NextRequest } from 'next/server';
import { createServerClient } from '@supabase/ssr';

export async function middleware(request: NextRequest) {
  let response = NextResponse.next({ request });

  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        getAll() {
          return request.cookies.getAll();
        },
        setAll(cookiesToSet, cacheHeaders) {
          for (const { name, value } of cookiesToSet) {
            request.cookies.set(name, value);
          }

          response = NextResponse.next({ request });

          for (const { name, value, options } of cookiesToSet) {
            response.cookies.set(name, value, options);
          }

          // Prevents a CDN from caching a response that carries Set-Cookie
          // and leaking one user's session to another.
          for (const [key, value] of Object.entries(cacheHeaders ?? {})) {
            if (value) response.headers.set(key, value);
          }
        },
      },
    },
  );

  // Verifies the JWT with Supabase Auth and triggers a refresh when needed.
  // Do not substitute getSession() here: it decodes the cookie without
  // verifying it, which would make this middleware decorative.
  await supabase.auth.getUser();

  // Add a redirect here when you introduce routes that require a session:
  //   if (!user && request.nextUrl.pathname.startsWith('/dashboard')) { ... }

  return response;
}

export const config = {
  matcher: ['/((?!api|_next/static|_next/image|favicon.ico|.*\\.(?:svg|png|jpg|jpeg|gif|webp|ico)$).*)'],
};
```

```ts
// ═══════════════════════════════════════════════════════════════════════════
//  app/auth/callback/route.ts
//  Completes the PKCE exchange for magic links and OAuth redirects.
// ═══════════════════════════════════════════════════════════════════════════
import { NextResponse, type NextRequest } from 'next/server';
import { createSupabaseServerClient } from '@/lib/supabase/server';

export const runtime = 'edge';
export const dynamic = 'force-dynamic';

export async function GET(request: NextRequest) {
  const { searchParams, origin } = new URL(request.url);
  const code = searchParams.get('code');
  const nextParam = searchParams.get('next') ?? '/';
  const next = nextParam.startsWith('/') ? nextParam : '/';
  const base = process.env.NEXT_PUBLIC_SITE_URL ?? origin;

  if (!code) {
    return NextResponse.redirect(`${base}/?auth=missing_code`);
  }

  const supabase = await createSupabaseServerClient();
  const { error } = await supabase.auth.exchangeCodeForSession(code);

  if (error) {
    return NextResponse.redirect(`${base}/?auth=exchange_failed`);
  }

  // The cookies written by exchangeCodeForSession are attached to this
  // response automatically by the Next.js cookie store.
  return NextResponse.redirect(`${base}${next}`);
}
```
