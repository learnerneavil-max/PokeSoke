-- ───────────────────────────── shared helpers ─────────────────────────────
create or replace function public.touch_updated_at()
returns trigger
language plpgsql
as $$
begin
  new.updated_at := now();
  return new;
end;
$$;

-- Deterministic XP→level curve. MIRRORED in lib/game/engine.ts (levelForXp).
-- xp 0→L1, 50→L2, 200→L3, 450→L4, 800→L5 …
create or replace function public.level_for_xp(p_xp integer)
returns integer
language sql
immutable
as $$
  select least(100, greatest(1, 1 + floor(sqrt(greatest(p_xp, 0)::numeric / 50))::int));
$$;

-- ═════════════════════════════════ 1. profiles ════════════════════════════
create table if not exists public.profiles (
  id           uuid primary key references auth.users (id) on delete cascade,
  handle       text unique,
  display_name text,
  avatar_url   text,
  coins        integer not null default 500 check (coins >= 0),
  wins         integer not null default 0,
  losses       integer not null default 0,
  created_at   timestamptz not null default now(),
  updated_at   timestamptz not null default now()
);

drop trigger if exists profiles_touch on public.profiles;
create trigger profiles_touch
  before update on public.profiles
  for each row execute function public.touch_updated_at();

-- Auto-provision a profile row for every new auth user.
create or replace function public.handle_new_user()
returns trigger
language plpgsql
security definer
set search_path = public, pg_temp
as $$
begin
  insert into public.profiles (id, display_name, avatar_url)
  values (
    new.id,
    coalesce(
      new.raw_user_meta_data ->> 'full_name',
      new.raw_user_meta_data ->> 'name',
      split_part(coalesce(new.email, 'player'), '@', 1)
    ),
    new.raw_user_meta_data ->> 'avatar_url'
  )
  on conflict (id) do nothing;
  return new;
end;
$$;

drop trigger if exists on_auth_user_created on auth.users;
create trigger on_auth_user_created
  after insert on auth.users
  for each row execute function public.handle_new_user();

-- ════════════════════════════════ 2. characters ═══════════════════════════
create table if not exists public.characters (
  id            uuid primary key default gen_random_uuid(),
  owner_id      uuid not null references public.profiles (id) on delete cascade,
  name          text not null check (char_length(name) between 2 and 40),
  tagline       text check (char_length(tagline) <= 160),
  avatar_url    text,
  archetype     text not null check (char_length(archetype) between 2 and 40),
  -- { bravery, wit, charm, chaos, empathy, resolve } each 0–100
  traits        jsonb not null default '{}'::jsonb,
  persona_prompt text not null default '',
  level         integer not null default 1 check (level between 1 and 100),
  xp            integer not null default 0 check (xp >= 0),
  mood          text not null default 'curious',
  energy        integer not null default 100 check (energy between 0 and 100),
  wins          integer not null default 0,
  losses        integer not null default 0,
  is_public     boolean not null default true,
  created_at    timestamptz not null default now(),
  updated_at    timestamptz not null default now()
);

create index if not exists characters_owner_idx on public.characters (owner_id);
create index if not exists characters_public_idx on public.characters (created_at desc) where is_public;

drop trigger if exists characters_touch on public.characters;
create trigger characters_touch
  before update on public.characters
  for each row execute function public.touch_updated_at();

-- ══════════════════════════ 4. memories ("learned") ═══════════════════════
create table if not exists public.memories (
  id           uuid primary key default gen_random_uuid(),
  character_id uuid not null references public.characters (id) on delete cascade,
  owner_id     uuid not null references public.profiles (id) on delete cascade,
  kind         text not null default 'fact'
               check (kind in ('fact', 'preference', 'relationship', 'event', 'emotion')),
  content      text not null check (char_length(content) between 4 and 400),

  created_at   timestamptz not null default now(),

create index if not exists memories_character_rank_idx

-- ═══════════════════════════════ 5. battles ═══════════════════════════════
create table if not exists public.battles (
  id              uuid primary key default gen_random_uuid(),
  challenger_id   uuid not null references public.characters (id) on delete cascade,
  defender_id     uuid not null references public.characters (id) on delete cascade,
  challenger_owner uuid not null references public.profiles (id) on delete cascade,
  defender_owner   uuid not null references public.profiles (id) on delete cascade,
  winner_id       uuid references public.characters (id) on delete set null,
  status          text not null default 'resolved'
                  check (status in ('pending', 'resolved', 'cancelled')),
  turns           jsonb not null default '[]'::jsonb,
  log             text[] not null default '{}',
  rounds          integer not null default 0,
  rng_seed        bigint not null default 0,
  xp_awarded      integer not null default 0,
  resolved_at     timestamptz,
  created_at      timestamptz not null default now(),
  constraint battles_distinct_fighters check (challenger_id <> defender_id)
);

create index if not exists battles_challenger_idx
  on public.battles (challenger_owner, created_at desc);
create index if not exists battles_defender_idx
  on public.battles (defender_owner, created_at desc);

-- ══════════════════ 6. battle resolution (atomic, server-only) ════════════
-- Both fighters' XP/level/W-L are mutated here rather than from the API route,
-- because the attacker may not be allowed to UPDATE the defender's row under
-- RLS. SECURITY DEFINER lets the function cross that boundary — but it
-- re-checks participation itself, so a caller can only ever settle a battle
-- they actually fought in, and only once.
create or replace function public.apply_battle_result(p_battle_id uuid)
returns jsonb
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare
  b         public.battles;
  v_caller  uuid := auth.uid();
  v_base    integer := 30;  -- participation XP for both fighters
  v_bonus   integer := 20;  -- extra XP for the winner
  v_total   integer;
begin
  select * into b from public.battles where id = p_battle_id;
  if not found then
    raise exception 'battle % not found', p_battle_id using errcode = 'no_data_found';
  end if;

  if v_caller is null or v_caller not in (b.challenger_owner, b.defender_owner) then
    raise exception 'caller is not a participant in battle %', p_battle_id
      using errcode = 'insufficient_privilege';
  end if;

  if b.status <> 'resolved' then
    raise exception 'battle % is not resolved', p_battle_id;
  end if;

  if b.xp_awarded > 0 then
    return jsonb_build_object('alreadyApplied', true, 'xpAwarded', b.xp_awarded,
                              'base', v_base, 'bonus', v_bonus);
  end if;

  v_total := v_base + v_bonus;

  -- Winner. SET expressions read pre-UPDATE column values, so `level` is
  -- derived from the *new* xp in the same statement.
  update public.characters
     set xp     = xp + v_total,
         level  = public.level_for_xp(xp + v_total),
         wins   = wins + 1,
         energy = greatest(15, energy - 20),
         mood   = (array['smug', 'inspired', 'fierce']::text[])[1 + floor(random() * 3)::int]
   where id = b.winner_id;

  -- Loser.
  update public.characters
     set xp     = xp + v_base,
         level  = public.level_for_xp(xp + v_base),
         losses = losses + 1,
         energy = greatest(10, energy - 30),
         mood   = (array['furious', 'melancholic', 'determined']::text[])[1 + floor(random() * 3)::int]
   where id in (b.challenger_id, b.defender_id)
     and id <> b.winner_id;

  -- Profile-level aggregate counters.
  update public.profiles set wins = wins + 1 where id = (
    case when b.winner_id = b.challenger_id then b.challenger_owner else b.defender_owner end
  );
  update public.profiles set losses = losses + 1 where id = (
    case when b.winner_id = b.challenger_id then b.defender_owner else b.challenger_owner end
  );

  update public.battles
     set xp_awarded = v_total, resolved_at = now()
   where id = b.id;

  return jsonb_build_object('alreadyApplied', false, 'xpAwarded', v_total,
                            'base', v_base, 'bonus', v_bonus);
end;
$$;

revoke all on function public.apply_battle_result(uuid) from public, anon;
grant execute on function public.apply_battle_result(uuid) to authenticated;

-- ═══════════════════════════════ 7. RLS ═══════════════════════════════════
alter table public.profiles   enable row level security;
alter table public.characters enable row level security;
alter table public.messages   enable row level security;
alter table public.memories   enable row level security;
alter table public.battles    enable row level security;

-- profiles: readable by any signed-in player (for leaderboards / handles),
-- writable only by the owner. `coins` and W/L are mutated by server RPCs only.
drop policy if exists "profiles readable by authenticated" on public.profiles;
create policy "profiles readable by authenticated"
  on public.profiles for select to authenticated using (true);

drop policy if exists "profiles self update" on public.profiles;
create policy "profiles self update"
  on public.profiles for update to authenticated
  using (auth.uid() = id) with check (auth.uid() = id);

drop policy if exists "profiles self insert" on public.profiles;
create policy "profiles self insert"
  on public.profiles for insert to authenticated with check (auth.uid() = id);

-- characters: you always see your own; everyone sees public ones (arena roster).
drop policy if exists "characters visible if public or owned" on public.characters;
create policy "characters visible if public or owned"
  on public.characters for select to authenticated
  using (is_public or owner_id = auth.uid());

drop policy if exists "characters owner insert" on public.characters;
create policy "characters owner insert"
  on public.characters for insert to authenticated with check (owner_id = auth.uid());

drop policy if exists "characters owner update" on public.characters;
create policy "characters owner update"
  on public.characters for update to authenticated
  using (owner_id = auth.uid()) with check (owner_id = auth.uid());

drop policy if exists "characters owner delete" on public.characters;
create policy "characters owner delete"
  on public.characters for delete to authenticated using (owner_id = auth.uid());

-- messages: the transcript is private. The AI writes them with the player's
-- own JWT, so `owner_id = auth.uid()` holds for both role values.
drop policy if exists "messages owner all" on public.messages;
create policy "messages owner all"
  on public.messages for all to authenticated
  using (owner_id = auth.uid()) with check (owner_id = auth.uid());

-- memories: strictly private, never exposed to the arena or to other players.
drop policy if exists "memories owner all" on public.memories;
create policy "memories owner all"
  on public.memories for all to authenticated
  using (owner_id = auth.uid()) with check (owner_id = auth.uid());

-- battles: visible only to the two participants. The challenger is the only
-- party allowed to insert (you initiate your own fights); mutation happens
-- through apply_battle_result, which is SECURITY DEFINER.
drop policy if exists "battles participants read" on public.battles;
create policy "battles participants read"
  on public.battles for select to authenticated
  using (auth.uid() = challenger_owner or auth.uid() = defender_owner);

drop policy if exists "battles challenger insert" on public.battles;
create policy "battles challenger insert"
  on public.battles for insert to authenticated
  with check (auth.uid() = challenger_owner);

-- ═══════════════════ 8. Realtime: publications + authorization ═══════════
do $$
begin
  if not exists (
    select 1 from pg_publication_tables
    where pubname = 'supabase_realtime' and tablename = 'battles'
  ) then
    alter publication supabase_realtime add table public.battles;
  end if;

  if not exists (
    select 1 from pg_publication_tables
    where pubname = 'supabase_realtime' and tablename = 'messages'
  ) then
    alter publication supabase_realtime add table public.messages;
  end if;

  if not exists (
    select 1 from pg_publication_tables
    where pubname = 'supabase_realtime' and tablename = 'characters'
  ) then
    alter publication supabase_realtime add table public.characters;
  end if;
end $$;

-- Private Broadcast channels are gated by RLS on realtime.messages. Topics are
-- namespaced by owner id:  user:<uuid>  ·  battle:<uuid>
-- (the battle route publishes one copy of each result to each participant's
--  own topic, so a player can never subscribe to another player's feed).
alter table realtime.messages enable row level security;

drop policy if exists "cf receive own topics" on realtime.messages;
create policy "cf receive own topics"
  on realtime.messages for select to authenticated
  using (
    realtime.topic() = 'user:'   || auth.uid()::text
    or realtime.topic() = 'battle:' || auth.uid()::text
  );

-- The battle route publishes with the secret key (which bypasses RLS). This
-- policy exists so an anon-key + user-JWT publisher also works if you swap
-- credentials later.
drop policy if exists "cf publish to own topics" on realtime.messages;
create policy "cf publish to own topics"
  on realtime.messages for insert to authenticated
  with check (
    realtime.topic() = 'user:'   || auth.uid()::text
    or realtime.topic() = 'battle:' || auth.uid()::text
  );
