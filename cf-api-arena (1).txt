// ═══════════════════════════════════════════════════════════════════════════
//  app/api/battle/route.ts
//
//  Server-authoritative combat. The client sends only two character ids; every
//  number in the result — damage, crits, XP, level-ups — is computed here. A
//  tampered client cannot fabricate a win, and the stored rng_seed makes any
//  dispute replayable into an identical log.
// ═══════════════════════════════════════════════════════════════════════════
import { NextResponse, type NextRequest } from 'next/server';
import { simulateBattle, XP_PARTICIPATION, XP_WIN_BONUS } from '@/lib/game/engine';
import { createSupabaseServerClient, createSupabaseTokenClient } from '@/lib/supabase/server';
import type { Battle, BattleOutcome, Character } from '@/lib/types';

export const runtime = 'edge';
export const dynamic = 'force-dynamic';
export const maxDuration = 30;

const json = (body: unknown, status = 200) =>
  NextResponse.json(body, { status, headers: { 'Cache-Control': 'no-store' } });

/**
 * Publishes a battle result to a participant's private Realtime topic.
 *
 * Deliberately HTTP, not a websocket: the Edge runtime has no long-lived socket,
 * and Supabase exposes a REST broadcast endpoint for exactly this case
 * (https://supabase.com/docs/guides/realtime/broadcast). The secret key is used
 * because the publisher is the trusted server; subscribers still need the
 * `realtime.messages` SELECT policy from the migration to hear it.
 *
 * Topics are namespaced per owner id, so a player can only ever subscribe to
 * their own feed — the RLS policy enforces that too.
 */
async function broadcastBattleResult(ownerId: string, outcome: BattleOutcome) {
  const url = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const secret = process.env.SUPABASE_SERVICE_ROLE_KEY;
  if (!url || !secret) return;

  const topic = `battle:${ownerId}`;

  try {
    const response = await fetch(
      `${url}/realtime/v1/api/broadcast/${encodeURIComponent(topic)}/events/battle.resolved?private=true`,
      {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          apikey: secret,
          Authorization: `Bearer ${secret}`,
        },
        body: JSON.stringify({
          battleId: outcome.battle.id,
          winnerId: outcome.battle.winner_id,
          challenger: { id: outcome.challenger.id, name: outcome.challenger.name },
          defender: { id: outcome.defender.id, name: outcome.defender.name },
          xpAwarded: outcome.xpAwarded,
          rounds: outcome.battle.rounds,
          at: new Date().toISOString(),
        }),
      },
    );

    if (!response.ok) {
      // Realtime delivery is a notification layer, not a source of truth — the
      // battle row is already committed, so this is logged, not fatal.
      console.error('[battle] broadcast rejected', response.status, await response.text());
    }
  } catch (error) {
    console.error('[battle] broadcast failed', error);
  }
}

// ─────────────────────────────────────────────────────────────────────────
//  GET /api/battle?characterId=…  →  recent battle log for the signed-in user
// ─────────────────────────────────────────────────────────────────────────
export async function GET(request: NextRequest) {
  const supabase = await createSupabaseServerClient();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return json({ error: 'Unauthorized' }, 401);

  const characterId = request.nextUrl.searchParams.get('characterId');

  let query = supabase.from('battles').select('*')
    .order('created_at', { ascending: false }).limit(30);

  if (characterId) {
    query = query.or(`challenger_id.eq.${characterId},defender_id.eq.${characterId}`);
  }

  const { data, error } = await query;
  if (error) return json({ error: error.message }, 500);

  return json({ battles: (data ?? []) as Battle[] });
}

// ─────────────────────────────────────────────────────────────────────────
//  POST /api/battle   { challengerId, defenderId }
// ─────────────────────────────────────────────────────────────────────────
export async function POST(request: NextRequest) {
  let payload: unknown;
  try {
    payload = await request.json();
  } catch {
    return json({ error: 'Request body must be JSON.' }, 400);
  }

  const { challengerId, defenderId } = (payload ?? {}) as {
    challengerId?: string; defenderId?: string;
  };

  if (!challengerId || !defenderId) {
    return json({ error: 'challengerId and defenderId are required.' }, 400);
  }
  if (challengerId === defenderId) {
    return json({ error: 'A character cannot fight itself.' }, 400);
  }

  const supabase = await createSupabaseServerClient();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return json({ error: 'Unauthorized' }, 401);

  const { data: { session } } = await supabase.auth.getSession();
  const accessToken = session?.access_token;
  if (!accessToken) return json({ error: 'Session expired. Please sign in again.' }, 401);

  // RLS limits this to public characters + the caller's own. If the defender is
  // private it simply will not come back — hence the undefined check below.
  const { data: rows, error: fetchError } = await supabase
    .from('characters').select('*').in('id', [challengerId, defenderId]);

  if (fetchError) return json({ error: fetchError.message }, 500);

  const challenger = (rows ?? []).find((r) => r.id === challengerId) as Character | undefined;
  const defender = (rows ?? []).find((r) => r.id === defenderId) as Character | undefined;

  if (!challenger || !defender) {
    return json({ error: 'One of the fighters could not be found or has disabled challenges.' }, 404);
  }
  if (challenger.owner_id !== user.id) {
    return json({ error: 'You can only send your own characters into the arena.' }, 403);
  }

  // ── Simulation. Seeded from the platform CSPRNG so results are unguessable
  //    but permanently reproducible from the stored value.
  const seed = crypto.getRandomValues(new Uint32Array(1))[0];
  const simulation = simulateBattle({ challenger, defender, seed });

  const battleId = crypto.randomUUID();

  const battleInsert: Battle = {
    id: battleId,
    challenger_id: challenger.id,
    defender_id: defender.id,
    challenger_owner: challenger.owner_id,
    defender_owner: defender.owner_id,
    winner_id: simulation.winnerId,
    status: 'resolved',
    turns: simulation.turns,
    log: simulation.log,
    rounds: simulation.rounds,
    rng_seed: seed,
    xp_awarded: 0, // filled by apply_battle_result
    resolved_at: null,
    created_at: new Date().toISOString(),
  };

  const { error: insertError } = await supabase.from('battles').insert(battleInsert);
  if (insertError) {
    console.error('[battle] insert failed', insertError);
    return json({ error: 'Could not record the battle.' }, 500);
  }

  // ── Settle XP, levels, W/L and moods atomically, as the caller (so the RPC's
  //    internal participant check applies). Both rows update inside a SECURITY
  //    DEFINER function because the challenger has no UPDATE rights on the
  //    defender's character under RLS.
  const db = createSupabaseTokenClient(accessToken);
  const { data: settlement, error: rpcError } = await db.rpc('apply_battle_result', {
    p_battle_id: battleId,
  });

  if (rpcError) {
    console.error('[battle] settlement failed', rpcError);
    // The battle row exists and is truthful; only progression failed. Report it
    // rather than pretending the fight never happened.
    return json({ error: 'Battle recorded but rewards could not be applied.' }, 500);
  }

  // ── Read back post-reward state for the UI.
  const { data: after } = await supabase
    .from('characters').select('*').in('id', [challenger.id, defender.id]);

  const settledChallenger = ((after ?? []).find((c) => c.id === challenger.id) ?? challenger) as Character;
  const settledDefender  = ((after ?? []).find((c) => c.id === defender.id) ?? defender) as Character;

  const levelUps = [
    { row: settledChallenger, previous: challenger },
    { row: settledDefender, previous: defender },
  ]
    .filter(({ row, previous }) => row.level > previous.level)
    .map(({ row, previous }) => ({
      characterId: row.id, name: row.name, from: previous.level, to: row.level,
    }));

  const storedBattle: Battle = {
    ...battleInsert,
    xp_awarded: settlement?.xpAwarded ?? XP_PARTICIPATION + XP_WIN_BONUS,
    resolved_at: new Date().toISOString(),
  };

  const outcome: BattleOutcome = {
    battle: storedBattle,
    challenger: settledChallenger,
    defender: settledDefender,
    xpAwarded: storedBattle.xp_awarded,
    levelUps,
  };

  // ── Notify both participants. The defender learns about the fight in real
  //    time even though only the challenger was present.
  const recipients = Array.from(new Set([challenger.owner_id, defender.owner_id]));
  await Promise.all(recipients.map((ownerId) => broadcastBattleResult(ownerId, outcome)));

  return json(outcome, 200);
}

// ═══════════════════════════════════════════════════════════════════════════
//  app/api/characters/route.ts
//  GET  — roster (mine) + public opponents for the arena.
//  POST — generate a character with the model, sanitise it, then persist it.
// ═══════════════════════════════════════════════════════════════════════════
import { NextResponse, type NextRequest } from 'next/server';
import OpenAI from 'openai';
import { buildCharacterDraftPrompt, UTILITY_MODEL } from '@/lib/ai/prompts';
import { createSupabaseServerClient } from '@/lib/supabase/server';
import {
  ARCHETYPES, DEFAULT_TRAITS, MOODS, TRAIT_KEYS,
  type Character, type TraitKey,
} from '@/lib/types';

export const runtime = 'edge';
export const dynamic = 'force-dynamic';

const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });

const MAX_CHARACTERS_PER_PLAYER = 6;
const MAX_BRIEF_CHARS = 400;

const json = (body: unknown, status = 200) =>
  NextResponse.json(body, { status, headers: { 'Cache-Control': 'no-store' } });

export async function GET() {
  const supabase = await createSupabaseServerClient();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return json({ error: 'Unauthorized' }, 401);

  const [mine, roster] = await Promise.all([
    supabase.from('characters').select('*')
      .eq('owner_id', user.id).order('created_at', { ascending: true }),
    // RLS on characters already filters this to is_public = true rows owned by
    // other players, so no extra predicate is needed here.
    supabase.from('characters').select('*')
      .neq('owner_id', user.id).order('level', { ascending: false }).limit(24),
  ]);

  return json({
    characters: (mine.data ?? []) as Character[],
    opponents: (roster.data ?? []) as Character[],
  });
}

interface DraftShape {
  name?: unknown; tagline?: unknown; archetype?: unknown;
  traits?: Record<string, unknown>; mood?: unknown; persona_prompt?: unknown;
}

function sanitiseTraits(input: DraftShape['traits']) {
  const traits = { ...DEFAULT_TRAITS };
  if (!input || typeof input !== 'object') return traits;

  for (const key of TRAIT_KEYS) {
    const raw = Number((input as Record<string, unknown>)[key as TraitKey]);
    if (Number.isFinite(raw)) traits[key] = Math.round(Math.min(100, Math.max(0, raw)));
  }
  return traits;
}

export async function POST(request: NextRequest) {
  let payload: unknown;
  try {
    payload = await request.json();
  } catch {
    return json({ error: 'Request body must be JSON.' }, 400);
  }

  const { brief, name, avatarUrl } = (payload ?? {}) as {
    brief?: string; name?: string; avatarUrl?: string;
  };

  const seedBrief = typeof brief === 'string' ? brief.trim().slice(0, MAX_BRIEF_CHARS) : '';
  if (seedBrief.length < 8) {
    return json({ error: 'Describe the character in at least a few words.' }, 400);
  }

  const supabase = await createSupabaseServerClient();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return json({ error: 'Unauthorized' }, 401);

  const { count } = await supabase.from('characters')
    .select('id', { count: 'exact', head: true }).eq('owner_id', user.id);

  if ((count ?? 0) >= MAX_CHARACTERS_PER_PLAYER) {
    return json(
      { error: `You are at the ${MAX_CHARACTERS_PER_PLAYER}-character limit. Retire one first.` },
      409,
    );
  }

  // ── Ask the model to design the persona.
  const { system, user: userPrompt } = buildCharacterDraftPrompt(seedBrief);

  let draft: DraftShape;
  try {
    const completion = await openai.chat.completions.create({
      model: UTILITY_MODEL,
      temperature: 0.95,
      max_tokens: 700,
      response_format: { type: 'json_object' },
      messages: [
        { role: 'system', content: system },
        { role: 'user', content: userPrompt },
      ],
    });
    draft = JSON.parse(completion.choices[0]?.message?.content ?? '{}') as DraftShape;
  } catch (error) {
    console.error('[characters] drafting failed', error);
    return json({ error: 'The forge is cold. Try again in a moment.' }, 502);
  }

  // ── Sanitise every field. Never persist unvalidated model output.
  const finalName =
    typeof name === 'string' && name.trim().length >= 2
      ? name.trim().slice(0, 40)
      : typeof draft.name === 'string' && draft.name.trim().length >= 2
        ? draft.name.trim().slice(0, 40)
        : null;

  if (!finalName) {
    return json({ error: 'Could not name that character. Try a different description.' }, 422);
  }

  const archetype =
    typeof draft.archetype === 'string' && (ARCHETYPES as readonly string[]).includes(draft.archetype)
      ? draft.archetype : 'rebel';

  const mood =
    typeof draft.mood === 'string' && (MOODS as readonly string[]).includes(draft.mood)
      ? draft.mood : 'curious';

  const personaPrompt =
    typeof draft.persona_prompt === 'string' && draft.persona_prompt.trim().length > 40
      ? draft.persona_prompt.trim().slice(0, 2000)
      : `Speaks with dry confidence. Guards their past. Warms up slowly to anyone who shows real curiosity about ${seedBrief.slice(0, 80)}.`;

  const tagline = typeof draft.tagline === 'string' ? draft.tagline.trim().slice(0, 160) : null;

  const { data: created, error: insertError } = await supabase
    .from('characters')
    .insert({
      owner_id: user.id,
      name: finalName,
      tagline,
      avatar_url: typeof avatarUrl === 'string' ? avatarUrl.slice(0, 500) : null,
      archetype,
      traits: sanitiseTraits(draft.traits),
      persona_prompt: personaPrompt,
      mood,
    })
    .select()
    .single();

  if (insertError || !created) {
    console.error('[characters] insert failed', insertError);
    return json({ error: 'Could not save that character.' }, 500);
  }

  return json({ character: created as Character }, 201);
}

// ═══════════════════════════════════════════════════════════════════════════
//  app/api/bootstrap/route.ts
//  One-shot hydrate for the dashboard. The dashboard is a Client Component so it
//  holds its own state; this returns everything in a single round-trip rather
//  than three parallel calls from the browser.
// ═══════════════════════════════════════════════════════════════════════════
import { NextResponse } from 'next/server';
import { createSupabaseServerClient } from '@/lib/supabase/server';
import type { BootstrapPayload, Battle, Character, Profile } from '@/lib/types';

export const runtime = 'edge';
export const dynamic = 'force-dynamic';

export async function GET() {
  const supabase = await createSupabaseServerClient();
  const { data: { user } } = await supabase.auth.getUser();

  const noStore = { headers: { 'Cache-Control': 'no-store' } };

  if (!user) {
    return NextResponse.json(
      { profile: null, characters: [], opponents: [], battles: [] } satisfies BootstrapPayload,
      noStore,
    );
  }

  const [profile, mine, roster, battles] = await Promise.all([
    supabase.from('profiles').select('*').eq('id', user.id).maybeSingle(),
    supabase.from('characters').select('*')
      .eq('owner_id', user.id).order('created_at', { ascending: true }),
    supabase.from('characters').select('*')
      .neq('owner_id', user.id).order('level', { ascending: false }).limit(24),
    supabase.from('battles').select('*')
      .order('created_at', { ascending: false }).limit(20),
  ]);

  const payload: BootstrapPayload = {
    profile: (profile.data ?? null) as Profile | null,
    characters: (mine.data ?? []) as Character[],
    opponents: (roster.data ?? []) as Character[],
    battles: (battles.data ?? []) as Battle[],
  };

  return NextResponse.json(payload, noStore);
}
